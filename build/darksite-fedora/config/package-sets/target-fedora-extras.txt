# target-fedora-extras.txt — Fedora-specific additions to the shared EL
# package sets. These are the packages needed on Fedora targets that either
# don't exist on EL or whose names differ enough that shared EL sets miss them.
#
# Most important: the full DKMS toolchain for building OpenZFS from source.
# OpenZFS upstream hasn't published a prebuilt release RPM for fc43 yet
# (as of 2026-04), so fedora installs compile ZFS from the kernel-devel
# headers at install-time. Without this chain, offline fedora installs
# would fail at the ZFS build step.

# DKMS module build system
dkms
kmod
kmodtool

# Autotools + make (ZFS ./configure needs autoconf 2.60+, automake 1.12+)
autoconf
automake
libtool
make
bc

# Compilers (gcc is pulled in by kernel-devel, but be explicit)
gcc
gcc-c++
cpp

# RPM building (ZFS configure --with rpm emits RPMs we can install)
rpm-build
rpmdevtools

# ZFS userspace + kernel build headers
libblkid-devel
libuuid-devel
# F44: no standalone libudev-devel — those headers ship in systemd-devel.
systemd-devel
libaio-devel
libattr-devel
libtirpc-devel
# F44 switched zlib -> zlib-ng; the -devel headers are in zlib-ng-compat-devel.
zlib-ng-compat-devel
libzstd-devel
openssl-devel
elfutils-libelf-devel

# Kernel build deps (kernel-devel already in shared set — add kernel-headers)
kernel-headers

# Python runtime + bindings ZFS userspace tools use
python3-devel
python3-cffi
python3-setuptools
python3-pip

# Perl is needed by ZFS's configure-time scripts and by dkms itself
perl-interpreter
perl-Getopt-Long
perl-File-Find

# Package manager — MUST be in the darksite so the install can put it on the
# target. The bootstrap.sh install array lists dnf5 explicitly (it was being
# --skip-broken'd away as a mere dep of the `dnf` compat metapackage, leaving
# the 10.100.10.113 F44 box with NO package manager). The post-install critical
# gate now hard-fails if dnf5 is absent, so it MUST resolve from this mirror.
dnf5
dnf5-plugins

# Editors — nano (terminal, universal) + gnome-text-editor (F44's GUI editor;
# gedit was dropped from F44 AppStream). Both must be in the mirror for the
# offline install to land an editor.
nano
gnome-text-editor

# Cert trust — nss-tools ships `certutil`, which kldload-trust-cert uses to
# seed Chrome's per-user NSS DB (~/.pki/nssdb) + the system store so the
# kldload-webui self-signed cert on https://localhost:8443 is trusted silently
# (no interstitial on every dashboard tile). It's already in the Debian darksite
# (target-base.txt) and the Fedora desktop install list (profiles.sh), and there
# is a firstboot healing net that re-attempts `dnf install nss-tools` — but on
# F44 that net FAILED because the .rpm was never mirrored here: only the repo
# *metadata* carried nss-tools (as a transitive dep), so the completeness gate's
# metadata dry-run passed while the actual file was absent (10.100.10.124
# 2026-06-14: "Could not open file .../nss-tools-3.123.1-1.fc44.x86_64.rpm",
# kldload-tls-cert.service then churned to start-limit-hit). Listing it here
# forces the .rpm into the snapshot so the offline install can land it.
nss-tools

# Firmware / tools that Fedora names differently vs EL
iotop-c

# Network-monitor TUIs — native in Fedora's repo but EPEL-only on EL, so they
# can't ride the shared EL package set and MUST be packed here for offline
# Fedora installs (the console cockpit binds Alt+n → nethogs, Alt+p → iftop).
# HISTORY: their absence made nethogs/iftop/iotop-c fail the air-gapped install
# on 10.100.10.136 (2026-07-05) — darksite file:// miss → network fallback →
# "all mirrors were tried" → firstboot WARN. iotop-c above is the same class.
nethogs
iftop

# Fedora's NATIVE default GNOME wallpaper (distinct day/night art). REQUIRED on
# the target: profiles.sh sets the per-distro native wallpaper at install
# (01-kldload-wallpaper → /usr/share/backgrounds/f44/default/f44-01-{day,night}
# .jxl) instead of the kldload nebula, so an F44 box looks like Fedora. The
# GNOME desktop group does NOT reliably drag this in on a minimal/offline
# install, so we list it explicitly — otherwise the wallpaper file is missing,
# the install logs a loud WARN, and the box boots with no wallpaper (F44 install
# on 10.100.10.113, 2026-06-13: came up with the wrong-distro nebula because the
# Fedora default art was never on the target). The completeness gate enforces
# its presence in the darksite.
f44-backgrounds-gnome

# Boolean/rich-dep companions that `dnf download --resolve --alldeps` does NOT
# follow (it ignores `(X if Y)` conditional deps). qemu pulls passt, and passt
# carries `Requires: (passt-selinux if selinux-policy-targeted)`. Since the
# target installs selinux-policy-targeted, that condition fires at install time
# and demands passt-selinux — which, if absent from the mirror, makes the ENTIRE
# offline dnf transaction unsatisfiable, taking the zfs pass-3 install down with
# it (F44 b-2026-06-13: install aborted "target will not boot" although zfs 2.4.3
# resolved fine). Listing it explicitly pulls it in the same snapshot as passt
# so the version matches. NOTE: if other rich-dep companions surface, add them
# here too, or build a post-mirror closure gate that dry-run-installs the full
# set against the darksite and reports gaps.
passt-selinux

# Fedora-grade desktop parity (1.3.0) — boot splash, GNOME utility apps, a11y,
# fingerprint, printer drivers, input methods + locales, and video-decode/GPU
# driver breadth. Bundled so the desktop comes up offline-complete.
plymouth
plymouth-system-theme
power-profiles-daemon
gnome-system-monitor
gnome-disk-utility
orca
at-spi2-core
speech-dispatcher
fprintd
libfprint
gutenprint
ibus
ibus-libpinyin
ibus-anthy
ibus-hangul
glibc-all-langpacks
# mesa-vdpau-drivers was dropped from F44 (no longer packaged) — removed.
libva-intel-driver
# F44: nvidia-vaapi-driver was renamed to libva-nvidia-driver.
libva-nvidia-driver
