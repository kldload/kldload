#
#   kldload.lab.example.com          ← DNS (no prefix)

#   - systemctl start kldload-tls-cert
#   - reboot
#   - hourly timer firing
#   - NetworkManager dispatcher event
#

#   systemctl try-restart kldload-tls-cert.service
